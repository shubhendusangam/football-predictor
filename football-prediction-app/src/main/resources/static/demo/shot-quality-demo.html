Usage Examples

1. Basic Usage (IIFE Version)

<!-- Include CSS --> <link rel="stylesheet" href="/css/shot-quality-card.css"> <!-- Include JS --> <script src="/js/shot-quality-card.js"></script> <!-- Container --> <div id="shot-quality-container"></div> <script> // Fetch and render a single card var container = document.getElementById('shot-quality-container'); ShotQualityCard.fetchAndRender(container, 'Arsenal', null); // Or render home/away pair ShotQualityCard.renderPair(container, 'Arsenal'); </script>

2. ES6 Module Usage

import { renderShotQualityCard, fetchAndRenderShotQualityCard } from './components/team/ShotQualityCard.js'; // With pre-fetched data const teamStats = { teamName: 'Arsenal', isHome: true, qualityScore: 7.5, // 0-10 scale shotAccuracy: 38.5, // percentage conversionRate: 0.32, // 0-1 scale shotsTrend: [ { shots: 15, goals: 3 }, { shots: 12, goals: 2 }, // ... last 10 matches ] }; renderShotQualityCard(container, teamStats); // Or fetch from API await fetchAndRenderShotQualityCard(container, 'Arsenal', true);

3. TeamAnalyticsPage Integration

import { createTeamAnalyticsPage, renderShotQualitySection } from './pages/TeamAnalyticsPage.js'; // Option 1: Full page const page = await createTeamAnalyticsPage('analytics-container', 'Arsenal'); // Option 2: Just the shot quality section await renderShotQualitySection(container, 'Arsenal');
